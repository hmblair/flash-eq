\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath,amssymb,mathtools}
\usetikzlibrary{
  arrows.meta, positioning, calc,
  decorations.pathreplacing, fit, backgrounds,
}

% ── Color palette ─────────────────────────────────────────────────
\definecolor{m0}{HTML}{5B9BD5}     % blue  (m=0)
\definecolor{m1}{HTML}{ED7D31}     % orange (m=1)
\definecolor{m2}{HTML}{70AD47}     % green  (m=2)
\definecolor{m3}{HTML}{C55A5A}     % muted red (m=3)
\definecolor{gry}{HTML}{B0B0B0}    % gray for ℓ-ordered features
\definecolor{tbl}{HTML}{7B68AE}    % purple for radial table
\definecolor{txt}{HTML}{2C2C2C}    % near-black text

% ── Dimensions ────────────────────────────────────────────────────
\def\cs{0.48}   % cell size
\def\tc{0.22}   % radial-table cell size

\begin{document}
\begin{tikzpicture}[
  >=Stealth,
  every node/.style={font=\small},
  step/.style={
    rounded corners=3pt, draw=txt!40, fill=txt!6,
    line width=0.4pt, inner sep=3pt 5pt,
    font=\footnotesize\bfseries, text=txt,
  },
  lbl/.style={font=\scriptsize, text=txt},
  sublbl/.style={font=\tiny, text=txt!60},
  arr/.style={->, thick, txt},
  bigarr/.style={-{Stealth[length=5pt,width=4pt]}, line width=1.2pt, txt!70},
]

% ── Computed lengths ──────────────────────────────────────────────
\pgfmathsetmacro{\W}{7*\cs}       % total width of 7-cell row
\pgfmathsetmacro{\C}{\W/2}        % center of row


% ══════════════════════════════════════════════════════════════════
%  STEP 1 — Rotate to m-basis
% ══════════════════════════════════════════════════════════════════

\node[step] at (-0.65, 0) {\strut 1};

% Input features f (ℓ-ordered, gray)
\begin{scope}[shift={(0, -0.25)}]
  \foreach \j in {0,...,6}
    {\fill[gry, opacity=0.25] (\j*\cs, 0) rectangle ++(\cs, \cs);}
  \draw[gray!50, thin] (0,0) grid[step=\cs] (\W, \cs);
  \draw[gray!70] (0,0) rectangle (\W, \cs);
  \node[lbl, above=3pt] at (\C, \cs)
    {$\mathbf{f}$ \;{\footnotesize($\ell$-ordered)}};
\end{scope}

% Arrow down
\draw[bigarr] (\C, -0.40) -- (\C, -1.10)
  node[midway, right=5pt, lbl] {$P(\hat{x})^{\!\top}$}
  node[midway, left=5pt, sublbl, align=right] {block-diag\\[-1pt]Wigner-$D$};

% Output features f̃ (m-ordered, colored)
\begin{scope}[shift={(0, -1.25)}]
  \fill[m0, opacity=0.50] (0,0) rectangle ++(\cs,\cs);
  \foreach \j in {0,1}{\fill[m1, opacity=0.50] ({(1+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \foreach \j in {0,1}{\fill[m2, opacity=0.50] ({(3+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \foreach \j in {0,1}{\fill[m3, opacity=0.50] ({(5+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \draw[gray!50, thin] (0,0) grid[step=\cs] (\W,\cs);
  \draw[gray!70] (0,0) rectangle (\W,\cs);
  \node[lbl, anchor=west] at ({\W+0.12}, {0.5*\cs})
    {$\tilde{\mathbf{f}}$ \;{\footnotesize($m$-ordered)}};
\end{scope}


% ══════════════════════════════════════════════════════════════════
%  STEP 2 — Block-diagonal multiply (fused, streamed from memory)
% ══════════════════════════════════════════════════════════════════

\def\STwo{-2.70}

\node[step] at (-0.65, {\STwo+0.15}) {\strut 2};

% ── Input vector f̃ (vertical, left of matrix) ───────────────────
\begin{scope}[shift={(-0.85, \STwo)}]
  \fill[m0, opacity=0.50] (0,0) rectangle ++(\cs,\cs);
  \foreach \i in {0,1}{\fill[m1, opacity=0.50] (0, {-(1+\i)*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,1}{\fill[m2, opacity=0.50] (0, {-(3+\i)*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,1}{\fill[m3, opacity=0.50] (0, {-(5+\i)*\cs}) rectangle ++(\cs,\cs);}
  \draw[gray!50, thin] (0,\cs) grid[step=\cs] (\cs, {-6*\cs});
  \draw[gray!70] (0,\cs) rectangle (\cs, {-6*\cs});
  \node[lbl, above=3pt] at (0.5*\cs, \cs) {$\tilde{\mathbf{f}}$};
\end{scope}

% × sign
\node[font=\normalsize, text=txt] at (-0.15, {\STwo-2.5*\cs}) {$\times$};

% ── Block-diagonal matrix Λ (only blocks drawn, no zero fill) ───
\begin{scope}[shift={(0.15, \STwo)}]
  % Faint dashed outline: the full matrix that is NEVER built
  \draw[gray!25, dashed, thin]
    (0, \cs) rectangle (\W, {-6*\cs});

  % m=0: 1×1
  \fill[m0, opacity=0.45] (0,0) rectangle ++(\cs,\cs);
  \draw[m0!50!black, semithick] (0,0) rectangle ++(\cs,\cs);
  \node[font=\tiny, m0!55!black] at (0.5*\cs, 0.5*\cs) {$\lambda_0$};

  % m=1: 2×2
  \fill[m1, opacity=0.45] (\cs, -\cs) rectangle ++(\cs,\cs);
  \node[font=\tiny, m1!55!black] at (1.5*\cs, -0.5*\cs) {$a_1$};
  \fill[m1, opacity=0.45] ({2*\cs}, -\cs) rectangle ++(\cs,\cs);
  \node[font=\tiny, m1!55!black] at (2.5*\cs, -0.5*\cs) {$b_1$};
  \fill[m1, opacity=0.45] (\cs, {-2*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m1!55!black] at (1.5*\cs, -1.5*\cs) {$\text{-}b_1$};
  \fill[m1, opacity=0.45] ({2*\cs}, {-2*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m1!55!black] at (2.5*\cs, -1.5*\cs) {$a_1$};
  \draw[m1!50!black, semithick] (\cs, 0) rectangle ++({2*\cs}, {-2*\cs});
  \draw[m1!35!black, thin] ({2*\cs}, 0) -- ({2*\cs}, {-2*\cs});
  \draw[m1!35!black, thin] (\cs, {-\cs}) -- ({3*\cs}, {-\cs});

  % m=2: 2×2
  \fill[m2, opacity=0.45] ({3*\cs}, {-3*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m2!55!black] at (3.5*\cs, -2.5*\cs) {$a_2$};
  \fill[m2, opacity=0.45] ({4*\cs}, {-3*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m2!55!black] at (4.5*\cs, -2.5*\cs) {$b_2$};
  \fill[m2, opacity=0.45] ({3*\cs}, {-4*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m2!55!black] at (3.5*\cs, -3.5*\cs) {$\text{-}b_2$};
  \fill[m2, opacity=0.45] ({4*\cs}, {-4*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m2!55!black] at (4.5*\cs, -3.5*\cs) {$a_2$};
  \draw[m2!50!black, semithick] ({3*\cs}, {-2*\cs}) rectangle ++({2*\cs}, {-2*\cs});
  \draw[m2!35!black, thin] ({4*\cs}, {-2*\cs}) -- ({4*\cs}, {-4*\cs});
  \draw[m2!35!black, thin] ({3*\cs}, {-3*\cs}) -- ({5*\cs}, {-3*\cs});

  % m=3: 2×2
  \fill[m3, opacity=0.45] ({5*\cs}, {-5*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m3!55!black] at (5.5*\cs, -4.5*\cs) {$a_3$};
  \fill[m3, opacity=0.45] ({6*\cs}, {-5*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m3!55!black] at (6.5*\cs, -4.5*\cs) {$b_3$};
  \fill[m3, opacity=0.45] ({5*\cs}, {-6*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m3!55!black] at (5.5*\cs, -5.5*\cs) {$\text{-}b_3$};
  \fill[m3, opacity=0.45] ({6*\cs}, {-6*\cs}) rectangle ++(\cs,\cs);
  \node[font=\tiny, m3!55!black] at (6.5*\cs, -5.5*\cs) {$a_3$};
  \draw[m3!50!black, semithick] ({5*\cs}, {-4*\cs}) rectangle ++({2*\cs}, {-2*\cs});
  \draw[m3!35!black, thin] ({6*\cs}, {-4*\cs}) -- ({6*\cs}, {-6*\cs});
  \draw[m3!35!black, thin] ({5*\cs}, {-5*\cs}) -- ({7*\cs}, {-5*\cs});

  % Label
  \node[lbl, above=3pt] at (\C, \cs) {$\Lambda(r)$};
\end{scope}

% = sign
\pgfmathsetmacro{\eqX}{0.15+\W+0.30}
\node[font=\normalsize, text=txt] at (\eqX, {\STwo-2.5*\cs}) {$=$};

% ── Output vector Λf̃ (vertical, right of matrix) ────────────────
\pgfmathsetmacro{\outX}{0.15+\W+0.55}
\begin{scope}[shift={(\outX, \STwo)}]
  \fill[m0, opacity=0.50] (0,0) rectangle ++(\cs,\cs);
  \foreach \i in {0,1}{\fill[m1, opacity=0.50] (0, {-(1+\i)*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,1}{\fill[m2, opacity=0.50] (0, {-(3+\i)*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,1}{\fill[m3, opacity=0.50] (0, {-(5+\i)*\cs}) rectangle ++(\cs,\cs);}
  \draw[gray!50, thin] (0,\cs) grid[step=\cs] (\cs, {-6*\cs});
  \draw[gray!70] (0,\cs) rectangle (\cs, {-6*\cs});
  \node[lbl, above=3pt] at (0.5*\cs, \cs) {$\Lambda\tilde{\mathbf{f}}$};
\end{scope}

% ── Radial lookup table (right side) ─────────────────────────────
\pgfmathsetmacro{\tblX}{\outX + \cs + 0.80}
\pgfmathsetmacro{\tblY}{\STwo - 1.40}
\begin{scope}[shift={(\tblX, \tblY)}]
  % Title
  \node[font=\scriptsize\bfseries, text=tbl!80!black, anchor=south west]
    at (0, {0.5*\tc+0.12}) {Radial lookup table $T$};
  % K bins × 7 entries
  \foreach \i in {0,...,5}{
    \foreach \j in {0,...,6}{
      \pgfmathsetmacro{\op}{20+18*sin(\i*50+\j*55)}
      \fill[tbl, opacity={\op/100}]
        (\j*\tc, -\i*\tc) rectangle ++(\tc,\tc);
    }
  }
  % Highlight interpolated row pair
  \draw[m1!80!black, semithick]
    (0, {-2*\tc}) rectangle ({7*\tc}, {-4*\tc});
  % Grid + border
  \draw[gray!25, thin] (0, \tc) grid[step=\tc] ({7*\tc}, {-5*\tc});
  \draw[gray!50] (0, \tc) rectangle ({7*\tc}, {-5*\tc});
  % Axis labels
  \node[font=\tiny, text=txt!65, rotate=90, anchor=south]
    at (-0.18, {-2*\tc}) {$K$ bins};
  % Distance pointer
  \draw[txt!65, semithick, -{Stealth[length=2pt]}]
    (-0.48, {-3*\tc}) -- (-0.14, {-3*\tc})
    node[midway, above, font=\tiny, text=txt!75] {$r$};
  % Interpolation annotation
  \node[font=\tiny, text=m1!75!black, anchor=west]
    at ({7*\tc+0.10}, {-3*\tc}) {interpolate};
\end{scope}

% ── Arrow: radial table streams weights to blocks ────────────────
% Route: from bottom of table, left to matrix right edge
\pgfmathsetmacro{\matRight}{0.15 + \W}
\pgfmathsetmacro{\tblBotY}{\tblY - 5*\tc}
\pgfmathsetmacro{\streamY}{\tblBotY - 0.25}

\draw[arr, tbl!65, line width=0.9pt, rounded corners=3pt]
  ({\tblX + 3.5*\tc}, \tblBotY)
  -- ({\tblX + 3.5*\tc}, \streamY)
  -- ({\matRight + 0.08}, \streamY)
  node[midway, below=2pt, font=\tiny, text=tbl!80!black] {stream weights};


% ══════════════════════════════════════════════════════════════════
%  STEP 3 — Rotate back to standard basis
% ══════════════════════════════════════════════════════════════════

\pgfmathsetmacro{\SThree}{\STwo - 7*\cs - 0.55}

\node[step] at (-0.65, {\SThree+0.25}) {\strut 3};

% m-ordered result Λf̃ (colored, contiguous)
\begin{scope}[shift={(0, \SThree)}]
  \fill[m0, opacity=0.50] (0,0) rectangle ++(\cs,\cs);
  \foreach \j in {0,1}{\fill[m1, opacity=0.50] ({(1+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \foreach \j in {0,1}{\fill[m2, opacity=0.50] ({(3+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \foreach \j in {0,1}{\fill[m3, opacity=0.50] ({(5+\j)*\cs},0) rectangle ++(\cs,\cs);}
  \draw[gray!50, thin] (0,0) grid[step=\cs] (\W,\cs);
  \draw[gray!70] (0,0) rectangle (\W,\cs);
  \node[lbl, above=3pt] at (\C, \cs)
    {$\Lambda\tilde{\mathbf{f}}$ \;{\footnotesize($m$-ordered)}};
\end{scope}

% Arrow down
\draw[bigarr] (\C, {\SThree-0.15}) -- (\C, {\SThree-0.80})
  node[midway, right=5pt, lbl] {$Q(\hat{x})$}
  node[midway, left=5pt, sublbl, align=right] {block-diag\\[-1pt]Wigner-$D$};

% Output features f̂ (gray, ℓ-ordered)
\begin{scope}[shift={(0, {\SThree-0.95})}]
  \foreach \j in {0,...,6}
    {\fill[gry, opacity=0.25] (\j*\cs, 0) rectangle ++(\cs, \cs);}
  \draw[gray!50, thin] (0,0) grid[step=\cs] (\W, \cs);
  \draw[gray!70] (0,0) rectangle (\W, \cs);
  \node[lbl, below=3pt] at (\C, 0)
    {$\hat{\mathbf{f}}$ \;{\footnotesize($\ell$-ordered)}};
\end{scope}

\end{tikzpicture}
\end{document}
