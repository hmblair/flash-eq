\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath,amssymb,mathtools}
\usetikzlibrary{arrows.meta, decorations.pathreplacing, patterns}

% ── Color palette (from cmuts/fig1/overview) ─────────────────────
\definecolor{m0}{HTML}{A44694}     % purple (m=0)
\definecolor{m1}{HTML}{F1A226}     % gold   (m=1)
\definecolor{m2}{HTML}{6DDB5C}     % green  (m=2)
\definecolor{txt}{HTML}{2C2C2C}    % near-black text

% ── Dimensions ────────────────────────────────────────────────────
\def\cs{0.58}   % cell size

\begin{document}
\begin{tikzpicture}[
  >=Stealth,
  every node/.style={font=\small},
  lbl/.style={font=\scriptsize, text=txt},
]

% L = {0,1,2}: p = 1+3+5 = 9
% m-ordered: m=0 (3 entries), m=1 (4 entries), m=2 (2 entries)
\def\N{9}  % matrix dimension
\pgfmathsetmacro{\W}{\N*\cs}
\pgfmathsetmacro{\C}{\W/2}

\def\matX{-0.55}
\pgfmathsetmacro{\matR}{\matX+\W}
\pgfmathsetmacro{\fvX}{\matR+0.55}
\pgfmathsetmacro{\outX}{\fvX+\cs+0.55}

% Radial weight table dimensions (3 summary blocks)
\pgfmathsetmacro{\matCenter}{\matX+\C}
\def\tblGap{0.15}  % gap between blocks
\def\bW{1.2}       % block width
\pgfmathsetmacro{\tblTotalW}{3*\bW+2*\tblGap}
\pgfmathsetmacro{\tblX}{\matCenter-\tblTotalW/2}
\pgfmathsetmacro{\tblY}{\cs+1.2}


% ── Horizontal distance axis with binned lookup ─────────────────
% Shows r landing between two bins on a distance axis; the two
% neighboring bins are highlighted and arrows fan down to Λ(r).
\def\binW{0.55}    % bin rectangle width
\def\binH{1.05}    % bin rectangle height (tall enough for proportional stripes)
\def\nBins{8}      % number of visible bins
\pgfmathsetmacro{\axisW}{\nBins*\binW}
\pgfmathsetmacro{\axisY}{\cs+1.4}          % y of bin bottoms
\pgfmathsetmacro{\axisX}{\matCenter-\axisW/2}  % left edge of axis

\begin{scope}[shift={(\axisX, \axisY)}]
  % Left ellipsis
  \node[font=\scriptsize, text=gray!60] at (-0.35, {0.5*\binH}) {$\cdots$};
  % Right ellipsis
  \node[font=\scriptsize, text=gray!60] at ({\axisW+0.35}, {0.5*\binH}) {$\cdots$};

  % Draw bins — highlight bins 3 and 4 (the two around r)
  \foreach \b in {0,...,7} {
    \pgfmathsetmacro{\bx}{\b*\binW}
    % Highlight bins k and k+1
    \ifnum\b=3
      \fill[gray!20] (\bx, 0) rectangle ++({\binW}, {\binH});
    \fi
    \ifnum\b=4
      \fill[gray!20] (\bx, 0) rectangle ++({\binW}, {\binH});
    \fi
    \draw[gray!50, thin] (\bx, 0) rectangle ++({\binW}, {\binH});
  }

  % Heavier outline on the two highlighted bins
  \pgfmathsetmacro{\hL}{3*\binW}
  \pgfmathsetmacro{\hR}{5*\binW}
  \draw[txt, semithick] (\hL, 0) rectangle (\hR, \binH);

  % Measurement line just above the bins
  \pgfmathsetmacro{\rPos}{3.7*\binW}  % slightly left of bin boundary
  \pgfmathsetmacro{\mY}{\binH+0.12}   % y of measurement line
  \pgfmathsetmacro{\sLen}{0.08}       % serif half-height
  \pgfmathsetmacro{\lblY}{\mY+0.12}   % fixed y for labels
  % Left segment: bin k left edge → r  (α)
  \draw[txt, thin] ({3*\binW}, {\mY-\sLen}) -- ({3*\binW}, {\mY+\sLen});
  \draw[txt, thin] ({3*\binW}, \mY) -- (\rPos, \mY);
  \draw[txt, thin] (\rPos, {\mY-\sLen}) -- (\rPos, {\mY+\sLen});
  \node[font=\tiny, text=txt, anchor=base] at ({(3*\binW+\rPos)/2}, \lblY) {$\alpha$};
  % Right segment: r → bin k+1 right edge  (1-α)
  \draw[txt, thin] ({5*\binW}, {\mY-\sLen}) -- ({5*\binW}, {\mY+\sLen});
  \draw[txt, thin] (\rPos, \mY) -- ({5*\binW}, \mY);
  \node[font=\tiny, text=txt, anchor=base] at ({(\rPos+5*\binW)/2}, \lblY) {$1\!-\!\alpha$};

  % r marker — triangle above the measurement line, pointing down
  \pgfmathsetmacro{\triBot}{\mY+0.22}
  \pgfmathsetmacro{\triTop}{\triBot+0.15}
  \fill[txt] (\rPos, \triBot) -- ({(\rPos)-0.07}, \triTop) -- ({(\rPos)+0.07}, \triTop) -- cycle;
  \node[font=\scriptsize, text=txt, above=1pt] at (\rPos, \triTop) {$r$};

  % Bin labels k and k+1 (below the bins)
  \node[font=\tiny, text=gray!60, below=2pt] at ({3.5*\binW}, 0) {$k$};
  \node[font=\tiny, text=gray!60, below=2pt] at ({4.5*\binW}, 0) {$k\!+\!1$};

  % ── Distance axis below bins ──────────────────────────────────
  \def\axBot{-0.38}  % y of axis line
  \def\tickH{0.06}   % tick half-height
  \draw[txt, thin, ->] (0, \axBot) -- ({\axisW+0.15}, \axBot);
  % Tick at left end only (right end has arrowhead)
  \draw[txt, thin] (0, {\axBot-\tickH}) -- (0, {\axBot+\tickH});
  % Labels
  \node[font=\tiny, text=txt, below=2pt] at (0, {\axBot-\tickH}) {$0$};
  \node[font=\tiny, text=txt, below=2pt] at (\axisW, {\axBot-\tickH}) {$r_{\mathrm{cut}}$};
  \node[font=\tiny, text=txt, below=2pt] at ({0.5*\axisW}, {\axBot-\tickH}) {interatomic distance};

  % Color-coded stripes — proportional to weight counts: m0=9, m1=8, m2=2
  \def\wTotal{19}
  \pgfmathsetmacro{\sM}{9*\binH/\wTotal}   % m=0 height
  \pgfmathsetmacro{\sMone}{8*\binH/\wTotal} % m=1 height
  \pgfmathsetmacro{\sMtwo}{2*\binH/\wTotal} % m=2 height
  % Bin k stripes (bottom to top: m0, m1, m2)
  \fill[m0, opacity=0.35] (\hL, 0) rectangle ++(\binW, \sM);
  \fill[m1, opacity=0.35] (\hL, \sM) rectangle ++(\binW, \sMone);
  \fill[m2, opacity=0.35] (\hL, {\sM+\sMone}) rectangle ++(\binW, \sMtwo);
  % Bin k+1 stripes
  \pgfmathsetmacro{\hM}{4*\binW}
  \fill[m0, opacity=0.35] (\hM, 0) rectangle ++(\binW, \sM);
  \fill[m1, opacity=0.35] (\hM, \sM) rectangle ++(\binW, \sMone);
  \fill[m2, opacity=0.35] (\hM, {\sM+\sMone}) rectangle ++(\binW, \sMtwo);

  % Table label (above-left of axis, avoiding r marker)
  \node[font=\normalsize, text=txt] at ({1.2*\binW}, {\binH+0.55}) {$\mathbf{T}$};
\end{scope}

% Arrows from highlighted bins down to matrix blocks
% Fan out from different x positions along the two-bin span
\pgfmathsetmacro{\arrY}{\axisY - 0.05}
\pgfmathsetmacro{\binLmid}{\axisX + 3.3*\binW}  % left third of span
\pgfmathsetmacro{\binCmid}{\axisX + 4.0*\binW}  % center of span
\pgfmathsetmacro{\binRmid}{\axisX + 4.7*\binW}  % right third of span
\def\arrPad{0.18}  % gap between arrowhead and block
\draw[->, thick, m0!70!black]
  (\binLmid, \arrY) to[out=-85, in=90, looseness=0.5]
  ({\matX + 1.5*\cs}, {\cs + \arrPad});
\draw[->, thick, m1!70!black]
  (\binCmid, \arrY) to[out=-80, in=90, looseness=0.45]
  ({\matX + 5*\cs}, {-2*\cs + \arrPad});
\draw[->, thick, m2!70!black]
  (\binRmid, \arrY) to[out=-70, in=90, looseness=0.45]
  ({\matX + 8*\cs}, {-6*\cs + \arrPad});

% ── Vertical f̃ (right of matrix) ────────────────────────────────
\begin{scope}[shift={(\fvX, 0)}]
  % m=0: 3 entries (ℓ = 0,1,2)
  \foreach \i in {0,1,2}{\fill[m0, opacity=0.50] (0, {-\i*\cs}) rectangle ++(\cs,\cs);}
  % m=±1: 4 entries (ℓ = 1,2 × ±1)
  \foreach \i in {0,...,3}{\fill[m1, opacity=0.50] (0, {-(3+\i)*\cs}) rectangle ++(\cs,\cs);}
  % m=±2: 2 entries (ℓ = 2 × ±2)
  \foreach \i in {0,1}{\fill[m2, opacity=0.50] (0, {-(7+\i)*\cs}) rectangle ++(\cs,\cs);}
  \draw[black, thin] (0,\cs) grid[step=\cs] (\cs, {-8*\cs});
  \draw[black] (0,\cs) rectangle (\cs, {-8*\cs});
  \node[font=\normalsize, text=txt, above=3pt] at (0.5*\cs, \cs) {$\tilde{\mathbf{f}}$};
  % Entry labels
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, 0.5*\cs)   {$\tilde{f}_{0}^{(0)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -0.5*\cs)  {$\tilde{f}_{0}^{(1)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -1.5*\cs)  {$\tilde{f}_{0}^{(2)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -2.5*\cs)  {$\tilde{f}_{1}^{(1)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -3.5*\cs)  {$\tilde{f}_{-1}^{(1)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -4.5*\cs)  {$\tilde{f}_{1}^{(2)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -5.5*\cs)  {$\tilde{f}_{-1}^{(2)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -6.5*\cs)  {$\tilde{f}_{2}^{(2)}$};
  \node[font=\small, scale=0.55, text=black] at (0.5*\cs, -7.5*\cs)  {$\tilde{f}_{-2}^{(2)}$};
\end{scope}

% × sign (centered between matrix right edge and f̃ left edge)
\node[font=\Large, text=txt] at ({(\matR+\fvX)/2}, {-3.5*\cs}) {$\times$};

% ── Block-diagonal matrix Λ ──────────────────────────────────────
\begin{scope}[shift={(\matX, 0)}]
  % Hatched background (zero regions, never materialized)
  \fill[pattern=north east lines, pattern color=gray!40, opacity=0.3]
    (0, \cs) rectangle (\W, {-8*\cs});
  % Dashed ghost outline
  \draw[gray!25, dashed, thin] (0, \cs) rectangle (\W, {-8*\cs});

  % ── m=0 block: 3×3 real matrix (rows 0–2, cols 0–2) ──────────
  \fill[white] (0, \cs) rectangle ({3*\cs}, {-2*\cs});
  \foreach \i in {0,1,2}{
    \foreach \j in {0,1,2}{
      \fill[m0, opacity=0.45] ({\j*\cs}, {-\i*\cs}) rectangle ++(\cs,\cs);
    }
  }
  \draw[black, semithick] (0, \cs) rectangle ++({3*\cs}, {-3*\cs});
  % Internal grid
  \foreach \j in {1,2}{\draw[m0!35!black, thin] ({\j*\cs}, \cs) -- ({\j*\cs}, {-2*\cs});}
  \foreach \i in {0,1}{\draw[m0!35!black, thin] (0, {-\i*\cs}) -- ({3*\cs}, {-\i*\cs});}
  % All Λ_{ℓ_in ℓ_out}^{(0)} entries (ℓ_in,ℓ_out ∈ {0,1,2})
  \node[font=\small, m0!55!black, scale=0.7] at (0.5*\cs, 0.5*\cs) {$\Lambda_{00}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (1.5*\cs, 0.5*\cs) {$\Lambda_{01}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (2.5*\cs, 0.5*\cs) {$\Lambda_{02}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (0.5*\cs, -0.5*\cs) {$\Lambda_{10}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (1.5*\cs, -0.5*\cs) {$\Lambda_{11}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (2.5*\cs, -0.5*\cs) {$\Lambda_{12}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (0.5*\cs, -1.5*\cs) {$\Lambda_{20}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (1.5*\cs, -1.5*\cs) {$\Lambda_{21}^{(0)}$};
  \node[font=\small, m0!55!black, scale=0.7] at (2.5*\cs, -1.5*\cs) {$\Lambda_{22}^{(0)}$};

  % ── m=1 block: 4×4 with [[A,B],[-B,A]] structure ─────────────
  % rows 3–6, cols 3–6
  % Fill four 2×2 sub-blocks (darker for A diagonal, lighter for B off-diagonal)
  \fill[white] ({3*\cs}, {-2*\cs}) rectangle ({7*\cs}, {-6*\cs});
  \fill[m1, opacity=0.45] ({3*\cs}, {-2*\cs}) rectangle ({5*\cs}, {-4*\cs});   % A (top-left)
  \fill[m1, opacity=0.30] ({5*\cs}, {-2*\cs}) rectangle ({7*\cs}, {-4*\cs});   % B (top-right)
  \fill[m1, opacity=0.30] ({3*\cs}, {-4*\cs}) rectangle ({5*\cs}, {-6*\cs});   % -B (bottom-left)
  \fill[m1, opacity=0.45] ({5*\cs}, {-4*\cs}) rectangle ({7*\cs}, {-6*\cs});   % A (bottom-right)
  % Outer border
  \draw[black, semithick] ({3*\cs}, {-2*\cs}) rectangle ++({4*\cs}, {-4*\cs});
  % Faint internal grid lines, skipping label centers
  % Vertical lines at x = 4, 6 (skip ±0.40 around label centers)
  \draw[m1!25!black, thin, opacity=0.4] ({4*\cs}, {-2*\cs}) -- ({4*\cs}, {-2.60*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({4*\cs}, {-3.40*\cs}) -- ({4*\cs}, {-4.60*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({4*\cs}, {-5.40*\cs}) -- ({4*\cs}, {-6*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({6*\cs}, {-2*\cs}) -- ({6*\cs}, {-2.60*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({6*\cs}, {-3.40*\cs}) -- ({6*\cs}, {-4.60*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({6*\cs}, {-5.40*\cs}) -- ({6*\cs}, {-6*\cs});
  % Horizontal lines at y = -3, -4, -5 (skip around centers)
  \draw[m1!25!black, thin, opacity=0.4] ({3*\cs}, {-3*\cs}) -- ({3.45*\cs}, {-3*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({4.55*\cs}, {-3*\cs}) -- ({5.45*\cs}, {-3*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({6.55*\cs}, {-3*\cs}) -- ({7*\cs}, {-3*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({3*\cs}, {-5*\cs}) -- ({3.45*\cs}, {-5*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({4.55*\cs}, {-5*\cs}) -- ({5.45*\cs}, {-5*\cs});
  \draw[m1!25!black, thin, opacity=0.4] ({6.55*\cs}, {-5*\cs}) -- ({7*\cs}, {-5*\cs});
  % Sub-block dividers (heavier dashed)
  \draw[m1!50!black, thin, densely dashed]
    ({5*\cs}, {-2*\cs}) -- ({5*\cs}, {-6*\cs});
  \draw[m1!50!black, thin, densely dashed]
    ({3*\cs}, {-4*\cs}) -- ({7*\cs}, {-4*\cs});
  % Centered sub-block labels
  \node[font=\small, text=black, scale=0.85] at (4*\cs, -3*\cs) {$\Lambda_{11}^{(1)}$};
  \node[font=\small, text=black, scale=0.85] at (6*\cs, -3*\cs) {$\Lambda_{12}^{(1)}$};
  \node[font=\small, text=black, scale=0.85] at (4*\cs, -5*\cs) {$\Lambda_{21}^{(1)}$};
  \node[font=\small, text=black, scale=0.85] at (6*\cs, -5*\cs) {$\Lambda_{22}^{(1)}$};

  % ── m=2 block: 2×2 rotation-scaling (rows 7–8, cols 7–8) ─────
  % Single filled block with centered label and faint grid (split around label)
  \fill[white] ({7*\cs}, {-6*\cs}) rectangle ++({2*\cs}, {-2*\cs});
  \fill[m2, opacity=0.45] ({7*\cs}, {-6*\cs}) rectangle ++({2*\cs}, {-2*\cs});
  \draw[black, semithick] ({7*\cs}, {-6*\cs}) rectangle ++({2*\cs}, {-2*\cs});
  % Vertical grid line, skipping ±0.40 around center
  \draw[m2!25!black, thin, opacity=0.4] ({8*\cs}, {-6*\cs}) -- ({8*\cs}, {-6.60*\cs});
  \draw[m2!25!black, thin, opacity=0.4] ({8*\cs}, {-7.40*\cs}) -- ({8*\cs}, {-8*\cs});
  % Horizontal grid line, skipping center
  \draw[m2!25!black, thin, opacity=0.4] ({7*\cs}, {-7*\cs}) -- ({7.45*\cs}, {-7*\cs});
  \draw[m2!25!black, thin, opacity=0.4] ({8.55*\cs}, {-7*\cs}) -- ({9*\cs}, {-7*\cs});
  \node[font=\small, text=black, scale=0.85] at (8*\cs, -7*\cs) {$\Lambda_{22}^{(2)}$};

  % Label
  \node[font=\normalsize, text=txt, above=3pt] at (\C, \cs) {$\Lambda(r)$};

\end{scope}

% = sign (centered between f̃ right edge and Λf̃ left edge)
\node[font=\Large, text=txt] at ({(\fvX+\cs+\outX)/2}, {-3.5*\cs}) {$=$};

% ── Vertical Λf̃ (right of matrix) ───────────────────────────────
\begin{scope}[shift={(\outX, 0)}]
  \foreach \i in {0,1,2}{\fill[m0, opacity=0.50] (0, {-\i*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,...,3}{\fill[m1, opacity=0.50] (0, {-(3+\i)*\cs}) rectangle ++(\cs,\cs);}
  \foreach \i in {0,1}{\fill[m2, opacity=0.50] (0, {-(7+\i)*\cs}) rectangle ++(\cs,\cs);}
  \draw[black, thin] (0,\cs) grid[step=\cs] (\cs, {-8*\cs});
  \draw[black] (0,\cs) rectangle (\cs, {-8*\cs});
  \node[font=\normalsize, text=txt, above=3pt] at (0.5*\cs, \cs) {$\Lambda(r)\tilde{\mathbf{f}}$};
\end{scope}
